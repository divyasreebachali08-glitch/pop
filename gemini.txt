    // Audio Synthesizer (Zero external file dependency)
    class SoundController {
      constructor() {
        this.ctx = null;
        this.muted = false;
      }
      init() {
        if (!this.ctx) {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (AudioCtx) this.ctx = new AudioCtx();
        }
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }
      toggleMute() {
        this.muted = !this.muted;
        return this.muted;
      }
      playJump() {
        if (this.muted) return;
        this.init();
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(160, now);
        osc.frequency.exponentialRampToValueAtTime(480, now + 0.16);
        gain.gain.setValueAtTime(0.25, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.16);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.16);
      }
      playEat() {
        if (this.muted) return;
        this.init();
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, now);
        osc.frequency.exponentialRampToValueAtTime(800, now + 0.1);
        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.1);
      }
      playFire() {
        if (this.muted) return;
        this.init();
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(600, now);
        osc.frequency.exponentialRampToValueAtTime(150, now + 0.2);
        gain.gain.setValueAtTime(0.3, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.2);
      }
      playSmash() {
        if (this.muted) return;
        this.init();
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'square';
        osc.frequency.setValueAtTime(120, now);
        osc.frequency.exponentialRampToValueAtTime(30, now + 0.25);
        gain.gain.setValueAtTime(0.3, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.25);
      }
    }

    // Main Game Engine
    class DinoGame {
      constructor() {
        this.canvas = document.getElementById('game-canvas');
        this.ctx = this.canvas.getContext('2d');
        this.audio = new SoundController();

        this.W = 1600;
        this.H = 900;
        this.groundY = 740;

        this.state = 'READY';
        this.nightMode = false;

        this.score = 0;
        this.distance = 0;
        this.meatCount = 0;
        this.rocksSmashed = 0;
        this.highScore = parseInt(localStorage.getItem('dino_escapade_hs') || '0', 10);

        this.baseSpeed = 9;
        this.speed = this.baseSpeed;

        this.dino = {
          x: 180,
          y: this.groundY,
          w: 120,
          h: 120,
          vy: 0,
          isGrounded: true,
          jumpCount: 0,
          maxJumps: 2,
          poweredUp: false,
          powerMeter: 0,
          powerTimer: 0
        };

        this.obstacles = [];
        this.collectibles = [];
        this.projectiles = [];
        this.obstacleTimer = 0;
        this.meatTimer = 0;

        this.bindUI();
        this.lastTime = performance.now();
        requestAnimationFrame((t) => this.loop(t));
      }

      bindUI() {
        document.getElementById('btn-start-game').addEventListener('click', () => this.startGame());
        document.getElementById('btn-pause').addEventListener('click', () => this.togglePause());
        document.getElementById('btn-resume').addEventListener('click', () => this.togglePause());
        document.getElementById('btn-pause-replay').addEventListener('click', () => this.startGame());
        document.getElementById('btn-go-replay').addEventListener('click', () => this.startGame());

        document.getElementById('btn-audio-toggle').addEventListener('click', (e) => {
          const muted = this.audio.toggleMute();
          e.target.textContent = muted ? '🔇' : '🔊';
        });

        document.getElementById('btn-theme-toggle').addEventListener('click', (e) => {
          this.nightMode = !this.nightMode;
          e.target.textContent = this.nightMode ? '🌙' : '☀️';
        });

        document.getElementById('btn-touch-jump').addEventListener('click', () => this.jump());
        document.getElementById('btn-touch-fire').addEventListener('click', () => this.shootFire());

        window.addEventListener('keydown', (e) => {
          if (this.state === 'READY' && (e.code === 'Space' || e.code === 'Enter')) {
            this.startGame();
          } else if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') {
            this.jump();
          } else if (e.code === 'KeyF') {
            this.shootFire();
          } else if (e.code === 'KeyP' || e.code === 'Escape') {
            this.togglePause();
          }
        });
      }

      startGame() {
        document.getElementById('loading-screen').classList.add('hidden');
        document.getElementById('pause-modal').classList.add('hidden');
        document.getElementById('gameover-modal').classList.add('hidden');
        this.canvas.classList.remove('grayscale-filter');

        this.score = 0;
        this.distance = 0;
        this.meatCount = 0;
        this.rocksSmashed = 0;
        this.speed = this.baseSpeed;

        this.dino.y = this.groundY;
        this.dino.vy = 0;
        this.dino.isGrounded = true;
        this.dino.jumpCount = 0;
        this.dino.poweredUp = false;
        this.dino.powerMeter = 0;

        this.obstacles = [];
        this.collectibles = [];
        this.projectiles = [];
        this.obstacleTimer = 60;
        this.meatTimer = 40;

        document.getElementById('btn-touch-fire').style.display = 'none';

        this.state = 'PLAYING';
      }

      togglePause() {
        if (this.state === 'PLAYING') {
          this.state = 'PAUSED';
          this.canvas.classList.add('grayscale-filter');
          document.getElementById('pause-modal').classList.remove('hidden');
        } else if (this.state === 'PAUSED') {
          this.state = 'PLAYING';
          this.canvas.classList.remove('grayscale-filter');
          document.getElementById('pause-modal').classList.add('hidden');
        }
      }

      gameOver() {
        this.state = 'GAMEOVER';
        this.canvas.classList.add('grayscale-filter');
        if (this.score > this.highScore) {
          this.highScore = this.score;
          localStorage.setItem('dino_escapade_hs', this.highScore.toString());
        }
        document.getElementById('go-score').textContent = this.score;
        document.getElementById('go-highscore').textContent = this.highScore;
        document.getElementById('go-meat').textContent = this.meatCount;
        document.getElementById('go-rocks').textContent = this.rocksSmashed;
        document.getElementById('gameover-modal').classList.remove('hidden');
      }

      jump() {
        if (this.state !== 'PLAYING') return;
        if (this.dino.isGrounded || this.dino.jumpCount < this.dino.maxJumps) {
          this.dino.vy = -20;
          this.dino.isGrounded = false;
          this.dino.jumpCount++;
          this.audio.playJump();
        }
      }

      shootFire() {
        if (this.state !== 'PLAYING' || !this.dino.poweredUp) return;
        this.projectiles.push({
          x: this.dino.x + this.dino.w,
          y: this.dino.y - this.dino.h / 2 - 10,
          w: 50,
          h: 25,
          vx: 22
        });
        this.audio.playFire();
      }

      update() {
        if (this.state !== 'PLAYING') return;

        this.speed = Math.min(22, this.baseSpeed + (this.distance / 300));
        this.distance += Math.floor(this.speed * 0.1);
        this.score += Math.floor(this.speed * 0.05);

        // Player movement
        this.dino.vy += 0.85;
        this.dino.y += this.dino.vy;
        if (this.dino.y >= this.groundY) {
          this.dino.y = this.groundY;
          this.dino.vy = 0;
          this.dino.isGrounded = true;
          this.dino.jumpCount = 0;
        }

        // Power-up duration check
        if (this.dino.poweredUp) {
          this.dino.powerTimer -= 16.6;
          if (this.dino.powerTimer <= 0) {
            this.dino.poweredUp = false;
            this.dino.powerMeter = 0;
            document.getElementById('btn-touch-fire').style.display = 'none';
          }
        }

        // Spawn obstacles
        this.obstacleTimer--;
        if (this.obstacleTimer <= 0) {
          this.obstacles.push({
            x: this.W + 50,
            y: this.groundY - 70,
            w: 60,
            h: 70
          });
          this.obstacleTimer = Math.floor(Math.random() * 40 + 70 - this.speed);
        }

        // Spawn meat
        this.meatTimer--;
        if (this.meatTimer <= 0) {
          this.collectibles.push({
            x: this.W + 50,
            y: this.groundY - 120 - (Math.random() * 100),
            w: 45,
            h: 45
          });
          this.meatTimer = Math.floor(Math.random() * 50 + 60);
        }

        // Obstacles collision & update
        for (let i = this.obstacles.length - 1; i >= 0; i--) {
          const obs = this.obstacles[i];
          obs.x -= this.speed;

          if (this.checkCollision(
            { x: this.dino.x + 20, y: this.dino.y - this.dino.h + 10, w: this.dino.w - 40, h: this.dino.h - 20 },
            obs
          )) {
            this.gameOver();
            return;
          }

          if (obs.x < -100) this.obstacles.splice(i, 1);
        }

        // Meat pickup
        for (let i = this.collectibles.length - 1; i >= 0; i--) {
          const meat = this.collectibles[i];
          meat.x -= this.speed;

          if (this.checkCollision(
            { x: this.dino.x, y: this.dino.y - this.dino.h, w: this.dino.w, h: this.dino.h },
            meat
          )) {
            this.audio.playEat();
            this.meatCount++;
            this.score += 100;
            this.collectibles.splice(i, 1);

            if (!this.dino.poweredUp) {
              this.dino.powerMeter += 25;
              if (this.dino.powerMeter >= 100) {
                this.dino.poweredUp = true;
                this.dino.powerTimer = 10000;
                document.getElementById('btn-touch-fire').style.display = 'block';
                const alert = document.getElementById('power-alert');
                alert.classList.add('show');
                setTimeout(() => alert.classList.remove('show'), 2000);
              }
            }
            continue;
          }

          if (meat.x < -100) this.collectibles.splice(i, 1);
        }

        // Projectiles logic
        for (let p = this.projectiles.length - 1; p >= 0; p--) {
          const proj = this.projectiles[p];
          proj.x += proj.vx;

          let hit = false;
          for (let o = this.obstacles.length - 1; o >= 0; o--) {
            if (this.checkCollision(proj, this.obstacles[o])) {
              this.audio.playSmash();
              this.rocksSmashed++;
              this.score += 250;
              this.obstacles.splice(o, 1);
              hit = true;
              break;
            }
          }

          if (hit || proj.x > this.W + 100) {
            this.projectiles.splice(p, 1);
          }
        }

        // Update UI
        document.getElementById('hud-score').textContent = this.score;
        document.getElementById('hud-distance').textContent = `${this.distance}m`;
        document.getElementById('hud-meat').textContent = this.meatCount;
      }

      checkCollision(r1, r2) {
        return (
          r1.x < r2.x + r2.w &&
          r1.x + r1.w > r2.x &&
          r1.y < r2.y + r2.h &&
          r1.y + r1.h > r2.y
        );
      }

      draw() {
        this.ctx.clearRect(0, 0, this.W, this.H);

        // Background
        this.ctx.fillStyle = this.nightMode ? '#1a252f' : '#f39c12';
        this.ctx.fillRect(0, 0, this.W, this.H);

        // Sun / Moon
        this.ctx.fillStyle = this.nightMode ? '#ecf0f1' : '#f1c40f';
        this.ctx.beginPath();
        this.ctx.arc(1300, 180, 70, 0, Math.PI * 2);
        this.ctx.fill();

        // Ground
        this.ctx.fillStyle = this.nightMode ? '#2c3e50' : '#d35400';
        this.ctx.fillRect(0, this.groundY, this.W, this.H - this.groundY);

        // Draw Dino (Vector Canvas rendering)
        this.ctx.save();
        this.ctx.translate(this.dino.x, this.dino.y - this.dino.h);
        if (this.dino.poweredUp) {
          this.ctx.shadowColor = '#e74c3c';
          this.ctx.shadowBlur = 20;
        }
        this.ctx.fillStyle = '#27ae60';
        this.ctx.fillRect(0, 0, this.dino.w, this.dino.h);
        // Eye
        this.ctx.fillStyle = '#fff';
        this.ctx.fillRect(80, 20, 20, 20);
        this.ctx.fillStyle = '#000';
        this.ctx.fillRect(90, 25, 10, 10);
        this.ctx.restore();

        // Draw Obstacles (Rocks)
        this.ctx.fillStyle = '#7f8c8d';
        this.obstacles.forEach(obs => {
          this.ctx.beginPath();
          this.ctx.moveTo(obs.x, obs.y + obs.h);
          this.ctx.lineTo(obs.x + obs.w / 2, obs.y);
          this.ctx.lineTo(obs.x + obs.w, obs.y + obs.h);
          this.ctx.closePath();
          this.ctx.fill();
        });

        // Draw Meat
        this.ctx.fillStyle = '#e74c3c';
        this.collectibles.forEach(meat => {
          this.ctx.beginPath();
          this.ctx.arc(meat.x + meat.w/2, meat.y + meat.h/2, meat.w/2, 0, Math.PI * 2);
          this.ctx.fill();
        });

        // Draw Fireballs
        this.ctx.fillStyle = '#e67e22';
        this.projectiles.forEach(p => {
          this.ctx.fillRect(p.x, p.y, p.w, p.h);
        });
      }

      loop(timestamp) {
        this.update();
        this.draw();
        requestAnimationFrame((t) => this.loop(t));
      }
    }

    window.addEventListener('DOMContentLoaded', () => {
      new DinoGame();
    });
  </script>
